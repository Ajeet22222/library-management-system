🔍 DBMS Concepts Demonstrated

INNER JOIN — Books with Authors JOIN 
SELECT b.title, CONCAT(a.first_name, ' ', a.last_name) AS author
FROM books b
INNER JOIN authors a ON b.author_id = a.author_id;
LEFT JOIN — All Authors with Books JOIN 
SELECT CONCAT(a.first_name, ' ', a.last_name) AS author, b.title
FROM authors a
LEFT JOIN books b ON a.author_id = b.author_id;
Multi-Table JOIN — Borrowing Details COMPLEX 
SELECT br.borrow_id, m.first_name, b.title, s.first_name AS staff
FROM borrowings br
JOIN members m ON br.member_id = m.member_id
JOIN books b ON br.book_id = b.book_id
JOIN staff s ON br.staff_id = s.staff_id;
Scalar Subquery — Books Above Average Price SUBQUERY 
SELECT title, price FROM books
WHERE price > (SELECT AVG(price) FROM books);
Correlated Subquery ADVANCED 
SELECT b.title, b.total_copies
FROM books b
WHERE b.total_copies < (
    SELECT AVG(b2.total_copies)
    FROM books b2
    WHERE b2.category_id = b.category_id
);
GROUP BY with HAVING AGGREGATE 
SELECT c.name, COUNT(b.book_id) AS book_count
FROM books b
JOIN categories c ON b.category_id = c.category_id
GROUP BY c.name
HAVING COUNT(b.book_id) > 2;
SUM, AVG, MIN, MAX AGGREGATE 
SELECT SUM(price * total_copies) AS total_value,
       AVG(price) AS avg_price,
       MIN(price) AS cheapest,
       MAX(price) AS costliest
FROM books;
View — Active Borrowings Report VIEW 
CREATE VIEW vw_active_borrowings AS
SELECT br.borrow_id, m.first_name, b.title,
       br.borrow_date, br.due_date,
       DATEDIFF(CURRENT_DATE, br.due_date) AS days_overdue
FROM borrowings br
JOIN members m ON br.member_id = m.member_id
JOIN books b ON br.book_id = b.book_id
WHERE br.return_date IS NULL;
Trigger — Auto Decrease Stock on Borrow TRIGGER 
CREATE TRIGGER trg_after_borrow_insert
AFTER INSERT ON borrowings
FOR EACH ROW
BEGIN
    UPDATE books
    SET available_copies = available_copies - 1
    WHERE book_id = NEW.book_id;
END;
Stored Procedure — Borrow a Book PROCEDURE 
CREATE PROCEDURE sp_borrow_book(
    IN p_book_id INT,
    IN p_member_id INT,
    IN p_staff_id INT
)
BEGIN
    DECLARE v_avail INT;
    SELECT available_copies INTO v_avail
    FROM books WHERE book_id = p_book_id;
    IF v_avail > 0 THEN
        INSERT INTO borrowings (...) VALUES (...);
    END IF;
END;

📝 Normalization Example — Why author info is in a separate table:
Before (Unnormalized): Books table had author_name, author_email, author_nationality → repeated for each book by same author (redundancy!)
After (3NF): Created separate authors table → books references via author_id (FK) → No redundancy ✅
